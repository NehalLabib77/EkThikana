rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function verified() {
      return signedIn() && request.auth.token.email_verified == true;
    }

    function userDoc() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid));
    }

    function isStudent() {
      return verified() && userDoc().data.role == 'student';
    }

    function isGroupMember(groupId) {
      return isStudent()
        && groupId is string
        && request.auth.uid in get(/databases/$(database)/documents/groups/$(groupId)).data.memberIds;
    }

    function canReadStudyDoc(data) {
      return isStudent() && (
        data.ownerId == request.auth.uid
        || data.visibility == 'public'
        || (data.visibility == 'group' && isGroupMember(data.groupId))
      );
    }

    function ownedCreate() {
      return verified() && request.resource.data.ownerId == request.auth.uid;
    }

    function ownedReadDelete() {
      return verified() && resource.data.ownerId == request.auth.uid;
    }

    function ownedUpdate() {
      return verified()
        && resource.data.ownerId == request.auth.uid
        && request.resource.data.ownerId == resource.data.ownerId;
    }

    match /users/{uid} {
      allow create: if signedIn()
        && request.auth.uid == uid
        && request.resource.data.role in ['student', 'general'];

      allow read: if verified() && request.auth.uid == uid;

      allow update: if verified()
        && request.auth.uid == uid
        && request.resource.data.role == resource.data.role;

      allow delete: if verified() && request.auth.uid == uid;

      match /saved_materials/{materialId} {
        allow read, write: if verified() && request.auth.uid == uid;
      }

      match /material_state/{materialId} {
        allow read, write: if verified() && request.auth.uid == uid;

        match /page_notes/{noteId} {
          allow read, write: if verified() && request.auth.uid == uid;
        }
      }
    }

    // ---------------- STUDENT-ONLY ACADEMIC DATA ----------------

    match /semesters/{id} {
      allow create: if isStudent() && request.resource.data.ownerId == request.auth.uid;
      allow read, delete: if isStudent() && resource.data.ownerId == request.auth.uid;
      allow update: if isStudent()
        && resource.data.ownerId == request.auth.uid
        && request.resource.data.ownerId == resource.data.ownerId;
    }

    match /subjects/{id} {
      allow create: if isStudent() && request.resource.data.ownerId == request.auth.uid;
      allow read, delete: if isStudent() && resource.data.ownerId == request.auth.uid;
      allow update: if isStudent()
        && resource.data.ownerId == request.auth.uid
        && request.resource.data.ownerId == resource.data.ownerId;
    }

    match /notes/{id} {
      allow create: if isStudent()
        && request.resource.data.ownerId == request.auth.uid
        && request.resource.data.visibility in ['private', 'group', 'public'];
      allow read: if canReadStudyDoc(resource.data);
      allow update, delete: if isStudent()
        && resource.data.ownerId == request.auth.uid
        && request.resource.data.ownerId == resource.data.ownerId;
    }

    match /materials/{id} {
      allow read: if canReadStudyDoc(resource.data);
      allow create, update, delete: if false; // trusted backend only
    }

    match /groups/{id} {
      allow read: if isStudent() && request.auth.uid in resource.data.memberIds;
      allow create, update, delete: if false; // trusted backend only
    }

    // ---------------- STUDENT + GENERAL PERSONAL DATA ----------------

    match /tasks/{id} {
      allow create: if ownedCreate();
      allow read, delete: if ownedReadDelete();
      allow update: if ownedUpdate();
    }

    match /medicines/{id} {
      allow create: if ownedCreate();
      allow read, delete: if ownedReadDelete();
      allow update: if ownedUpdate();
    }

    match /medicine_doses/{id} {
      allow create: if ownedCreate()
        && request.resource.data.status in ['pending', 'taken', 'skipped', 'missed'];
      allow read, delete: if ownedReadDelete();
      allow update: if ownedUpdate()
        && request.resource.data.status in ['pending', 'taken', 'skipped', 'missed'];
    }

    match /bazar_items/{id} {
      allow create: if ownedCreate();
      allow read, delete: if ownedReadDelete();
      allow update: if ownedUpdate();
    }

    match /daily_expenses/{id} {
      allow create: if ownedCreate() && request.resource.data.amount is number
        && request.resource.data.amount > 0;
      allow read, delete: if ownedReadDelete();
      allow update: if ownedUpdate()
        && request.resource.data.amount is number
        && request.resource.data.amount > 0;
    }

    match /commute_trips/{id} {
      allow create: if ownedCreate()
        && request.resource.data.actualFare is number
        && request.resource.data.actualFare > 0;
      allow read, delete: if ownedReadDelete();
      allow update: if ownedUpdate()
        && request.resource.data.actualFare is number
        && request.resource.data.actualFare > 0;
    }

    // Central expense ledger. IDs are deterministic by source + sourceRecordId.
    match /financial_transactions/{id} {
      allow create: if verified()
        && request.resource.data.ownerId == request.auth.uid
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.type == 'expense'
        && request.resource.data.source in ['daily', 'bazar', 'medicine', 'commute']
        && request.resource.data.sourceRecordId is string
        && request.resource.data.amount is number
        && request.resource.data.amount >= 0;

      allow read, delete: if verified() && resource.data.ownerId == request.auth.uid;

      allow update: if verified()
        && resource.data.ownerId == request.auth.uid
        && request.resource.data.ownerId == resource.data.ownerId
        && request.resource.data.userId == resource.data.userId
        && request.resource.data.type == 'expense'
        && request.resource.data.source in ['daily', 'bazar', 'medicine', 'commute']
        && request.resource.data.sourceRecordId == resource.data.sourceRecordId
        && request.resource.data.amount is number
        && request.resource.data.amount >= 0;
    }

    // Backend-only metering/moderation.
    match /ai_usage/{id} {
      allow read, write: if false;
    }

    match /upload_usage/{id} {
      allow read, write: if false;
    }

    match /reports/{id} {
      allow read, write: if false;
    }

    // Legacy LifeHub/finance collections (FamilyHub/RentMate/Wellness/old grocery/location/savings
    // stores) intentionally have no match rules and are therefore denied.
  }
}
